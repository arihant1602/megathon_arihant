cmake_minimum_required(VERSION 3.22)
project(incident_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
# Snapdragon 8 Elite (Oryon, ARMv8.7): FP16 arithmetic, dot product and int8 matmul. Without this ggml builds for
# baseline ARMv8.0 and F16/Q8 matmuls fall back to scalar code (minutes per Whisper call instead of ~0.3 s).
set(GGML_CPU_ARM_ARCH "armv8.2-a+fp16+dotprod+i8mm" CACHE STRING "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../../third_party/llama.cpp" llama-build)
add_library(incident_llama SHARED llama_jni.cpp)
target_link_libraries(incident_llama PRIVATE llama log)
target_link_options(incident_llama PRIVATE "-Wl,-Bsymbolic" "-Wl,--exclude-libs,ALL")
