cmake_minimum_required(VERSION 3.22)
project(incident_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
# Snapdragon 8 Elite (Oryon, ARMv8.7): FP16 arithmetic, dot product and int8 matmul. Without this ggml builds for
# baseline ARMv8.0 and F16/Q8 matmuls fall back to scalar code (minutes per Whisper call instead of ~0.3 s).
set(GGML_CPU_ARM_ARCH "armv8.2-a+fp16+dotprod+i8mm" CACHE STRING "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../../third_party/whisper.cpp" whisper-build)
# Whisper's encoder on the NPU: swap whisper.cpp's (Apple-only) external-encoder stub for the SNPE one.
if(SNPE_ROOT)
    set(WHISPER_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../third_party/whisper.cpp/src")
    get_target_property(whisper_sources whisper SOURCES)
    list(FILTER whisper_sources EXCLUDE REGEX "aneforge/whisper-aneforge\\.cpp$")
    list(APPEND whisper_sources "${CMAKE_CURRENT_SOURCE_DIR}/whisper_snpe_encoder.cpp")
    set_property(TARGET whisper PROPERTY SOURCES ${whisper_sources})
    add_library(SNPE SHARED IMPORTED)
    set_target_properties(SNPE PROPERTIES IMPORTED_LOCATION "${SNPE_ROOT}/lib/aarch64-android/libSNPE.so")
    target_include_directories(whisper PRIVATE "${SNPE_ROOT}/include/SNPE" "${WHISPER_SRC}")
    target_link_libraries(whisper PRIVATE SNPE log)
endif()
add_library(incident_whisper SHARED whisper_jni.cpp)
target_link_libraries(incident_whisper PRIVATE whisper log)
target_link_options(incident_whisper PRIVATE "-Wl,-Bsymbolic" "-Wl,--exclude-libs,ALL")
